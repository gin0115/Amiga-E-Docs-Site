Are you doing this several times?
Then you will always copy the string into Address mystr, which will
be inserted for every listitem, so all items will contain the same text.

and MUIA_List_DestructHook, MUIV_List_DestructHook_String
for this list to avoid this.

> doMethodA(mygad,[MUIM_List_InsertSingle,x,MUIV_List_Insert_Bottom])
>
> ...works fine.

Now, X always points to another memory location, while mystr is always
the same memory location.

Hi!

> It seems like MUI-listviews does not like E-strings.

No, that's not the problem.

> Every time I try to use an E-string in a
> listview, I get strange results. Initialy everything looks fine,
> but when you resize the window or use the scroller on the side,
> strange things start happening.

That is, because initialy (immediately after inserting) the string
still exists, but later (when MUI needs it again to redraw it) the
string is disposed (see below).

> Like this...
>
> DEF mystr[10]:STRING
> StrCopy(mystr,'Hello')
> doMethodA(mygad,[MUIM_List_InsertSingle,mystr,MUIV_List_Insert_Bottom])

In this case, mystr is a pointer to some memory which is large enough
to contain up to 10 characters. This memory (the STRING) is local to
the function and is therefore allocated on the stack. And as soon as
the function ends, the memeory (the STRING) is disposed!

By inserting the string into the list, you only insert the pointer to
the string into the list. A pointer that points to a string which
exists only as long as the function runs! By default MUI does not
copy the string itself, but only the pointer. You have to tell MUI
to create a copy of the string with:

     MUIA_List_ConstructHook, MUIV_List_ConstructHook_String,
     MUIA_List_DestructHook,  MUIV_List_DestructHook_String

> ...but..
>
> DEF x
> X:='Hello'
> doMethodA(mygad,[MUIM_List_InsertSingle,x,MUIV_List_Insert_Bottom])
>
> ...works fine.

Yes, because in this case the string is a constant string which is
part of the program. Therefore it is not disposed at the end of
the function!

As EC does not support FPU ASM, you have to use an external assembler.
Write a xx.asm file, assemble it to xx.o, use o2m and use the module.
Beware that EC pushes arguments from left to right.

bla_iii:
=09fmove.s a(a7),fp0
=09fsub.s b(a7),fp0
=09fmove.s fp0,([c_ptr,a7])
=09rts

The second statement: you are trying to multiply a floating point number
with a integer constant!

=09fmove.s a(a7),fp0
=09fadd.s fp0,fp0
=09fmove.l fp,([b_ptr,a7])
=09rts

/* Sends an ARexx command (Hide) to YAM. */

MODULE 'exec/lists',
       'exec/nodes',
       'exec/ports',
       'rexxsyslib',
       'rexx/storage'

DEF g_mport:PTR TO mp

PROC main()
  DEF rxmsg:PTR TO rexxmsg, ok, done=FALSE

  rexxsysbase:=OpenLibrary('rexxsyslib.library', 0)

  /* Create a msg port. */
  g_mport:=CreateMsgPort()
  g_mport.ln.name:='TESTPORT'
  g_mport.ln.pri:=0
  AddPort(g_mport)

  /* Send an ARexx command, "Hide", to YAM (YAM should be running). */
  ok:=rxcmdtoport('YAM', 'Hide')

  IF ok
    /* Wait for the ARexx reply message, then quit. */
    REPEAT
      WaitPort(g_mport)
      rxmsg:=GetMsg(g_mport)
      IF rxmsg.mn.ln.type = NT_REPLYMSG
        WriteF('ARexx Reply.\n')
        ClearRexxMsg(rxmsg, 16)  -> Free the ArgString(s).
        DeleteRexxMsg(rxmsg)     -> Then kill the msg.
        done:=TRUE
      ELSE
        WriteF('Someone else is talking to us?!\n')
        ReplyMsg(rxmsg)
      ENDIF
    UNTIL done
  ENDIF

  /* Kill our msg port. */
  RemPort(g_mport)
  WHILE (rxmsg:=GetMsg(g_mport)) DO ReplyMsg(rxmsg)
  DeleteMsgPort(g_mport)

  CloseLibrary(rexxsysbase)
ENDPROC

PROC rxcmdtoport(portname, cmd)
  DEF rxmsg:PTR TO rexxmsg, argstr, port=NIL

  /* Create/init. a Rexx Msg. */
  rxmsg:=CreateRexxMsg(g_mport, NIL, 'TESTPORT')
  argstr:=CreateArgstring(cmd, StrLen(cmd))
  rxmsg.args[0]:=argstr
  rxmsg.action:=RXCOMM

  /* Send the msg. */
  Forbid()
    IF (port:=FindPort(portname))
      PutMsg(port, rxmsg)
    ENDIF
  Permit()

  IF port = NIL
    -> Couldn't find <port>; kill the Argstr & the RexxMsg.
    DeleteArgstring(argstr)
    DeleteRexxMsg(rxmsg)
  ENDIF
ENDPROC port <> NIL

E will *allways* replace your stack. Even if the system stack is larger
than the stack E allocates..

The only thing you can do is make it system friendly.

PROC fixstack()
  DEF task : PTR TO tc , sp

  task := FindTask( NIL )
  MOVE.L A7,sp
  task.spupper := sp
  task.splower := sp - FreeStack()

ENDPROC

This will still leave task.spupper incorrect, but that shouldn't be a
problem, the stack just looks slightly smaller than it is.
You could add an aproximation of used stack to the value. If you have used
the OPT STACK=X then you can use
  task.spupper := sp + X - FreeStack()

/*
  Generate a table for a byte-wise 32-bit CRC calculation on the polynomial:
  x^32+x^26+x^23+x^22+x^16+x^12+x^11+x^10+x^8+x^7+x^5+x^4+x^2+x+1.

  Polynomials over GF(2) are represented in binary, one bit per coefficient,
  with the lowest powers in the most significant bit.  Then adding
polynomials
  is just exclusive-or, and multiplying a polynomial by x is a right shift
by
  one.  If we call the above polynomial p, and represent a byte as the
  polynomial q, also with the lowest power in the most significant bit (so
the
  byte 0xb1 is the polynomial x^7+x^3+x+1), then the CRC is (q*x^32) mod p,
  where a mod b means the remainder after dividing a by b.

  This calculation is done using the shift-register method of multiplying
and
  taking the remainder.  The register is initialized to zero, and for each
  incoming bit, x^32 is added mod p to the register if the bit is a one
(where
  x^32 mod p is p+x^32 = x^26+...+1), and the register is multiplied mod p
by
  x (which is shifting right by one and adding x^32 mod p if the bit shifted
  out is a one).  We start with the highest power (least significant bit) of
  q and repeat for all eight bits of q.

  The table is simply the CRC of all possible eight bit values.  This is all
  the information needed to generate CRC's on data a byte at a time for all
  combinations of CRC register values and incoming bytes.
*/

{
  uLong c;
  int n, k;
  uLong poly;            /* polynomial exclusive-or pattern */
  /* terms of polynomial defining this crc (except x^32): */
  static Byte p[] = {0,1,2,4,5,7,8,10,11,12,16,22,23,26};

  /* make exclusive-or pattern from polynomial (0xedb88320L) */
  poly = 0L;
  for (n = 0; n < sizeof(p)/sizeof(Byte); n++)
    poly |= 1L << (31 - p[n]);

  for (n = 0; n < 256; n++)
  {
    c = (uLong)n;
    for (k = 0; k < 8; k++)
      c = c & 1 ? poly ^ (c >> 1) : c >> 1;
    crc_table[n] = c;
  }

#define DO1(buf) crc = crc_table[((int)crc ^ (*buf++)) & 0xff] ^ (crc >> 8);
#define DO2(buf)  DO1(buf); DO1(buf);
#define DO4(buf)  DO2(buf); DO2(buf);
#define DO8(buf)  DO4(buf); DO4(buf);

*/
uLong crc32(crc, buf, len)
    uLong crc;
    const Bytef *buf;
    uInt len;
{
    if (buf == Z_NULL) return 0L;
#ifdef DYNAMIC_CRC_TABLE
    if (crc_table_empty)
      make_crc_table();
#endif
    crc = crc ^ 0xffffffffL;
    while (len >= 8)
    {
      DO8(buf);
      len -= 8;
    }
    if (len) do {
      DO1(buf);
    } while (--len);
    return crc ^ 0xffffffffL;
}

   command

ENDPROC

PROC openELib(name, path)
   DEF port, s[100]:STRING, em:elibmess
   port:=FindPort(name)
   IF port=NIL
      StringF(s, 'RUN >NIL: ')
      IF path THEN StrAdd(s, path, ALL)
      StrAdd(s, name, ALL)
      SystemTagList(s, NIL)
      Delay(10)
      port:=FindPort(name)
   ENDIF
   IF port
      sendCmndToELibPort(em, port, "OPEN")
   ELSE
      RETURN NIL
   ENDIF
ENDPROC em.base

PROC closeELib(base:PTR TO base)
   DEF em:elibmess
   sendCmndToELibPort(em, base.elib_port, "CLSE")
ENDPROC

PROC getELibProc(base:PTR TO base, name)
   DEF fl:PTR TO LONG
   fl:=base.proclist
   WHILE fl[]<>NIL
      IF StrCmp(fl[], name, ALL)
         fl++
         RETURN fl[] -> ptr to  proc
      ENDIF
      fl++
      fl++
   ENDWHILE
ENDPROC NIL

PROC becomeAnELib(proclist:PTR TO LONG, name)
   DEF base:base
   base.elib_port:=makeELibPort(name)
   base.proclist:=proclist
   elibMainLoop(base)
   unMakeELibPort(base.elib_port)
ENDPROC

/* Private Functions Here */

PROC sendCmndToELibPort(em:PTR TO elibmess, port, command)
      em.mn.ln.type:=NT_MESSAGE
      em.mn.length:=SIZEOF elibmess
      em.mn.replyport:=CreateMsgPort()
      em.command:=command
      PutMsg(port, em)
      WaitPort(em.mn.replyport)
      GetMsg(em.mn.replyport)
      DeleteMsgPort(em.mn.replyport)
ENDPROC

PROC makeELibPort(name)
   DEF elib_port:PTR TO mp
   elib_port:=CreateMsgPort()
   elib_port.ln.name:=name
   elib_port.ln.pri:=0
   AddPort(elib_port)
ENDPROC elib_port

PROC elibMainLoop(base:PTR TO base)
   DEF mpsig, exit, msg:PTR TO elibmess,
       elib_port:PTR TO mp, opencount,
       command

   opencount:=NIL
   elib_port:=base.elib_port
   mpsig:=Shl(1, elib_port.sigbit)
   exit:=FALSE
   REPEAT
      Wait(mpsig)
      WHILE (msg:=GetMsg(elib_port))<>NIL
         command:=msg.command
         IF command = "CLSE"
            opencount--
            IF opencount < 2 THEN exit:=TRUE
         ENDIF
         IF command = "OPEN"
            opencount++
            msg.base:=base
         ENDIF
         ReplyMsg(msg)
      ENDWHILE
   UNTIL exit
ENDPROC

PROC unMakeELibPort(elib_port)
   RemPort(elib_port)
   DeleteMsgPort(elib_port)
ENDPROC
