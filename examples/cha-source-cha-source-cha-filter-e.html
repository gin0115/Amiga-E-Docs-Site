/*==========================================================================+
| cha_filter.e                                                              |
| various filters for OctaMED SoundStudio                                   |
|                                                                           |
| main program                                                              |
|                                                                           |
| Based on mkfilter/mkshape/genplot:                                        |
|   A. J. Fisher                                                            |
|   fisher@minster.york.ac.uk                                               |
+--------------------------------------------------------------------------*/

OPT PREPROCESS, LARGE, OSVERSION=37

-> experimental GUI, not for release in this version
-> #define PLOT

MODULE '*filter_splane', '*filter_zplane', '*filter',
       '*oss_input', '*oss_output', '*oss_sample', '*fxfilter', '*oss',
       '*filter_specification', '*errors'
#ifdef PLOT
MODULE '*filter_plot', 'tools/easygui'
#endif

RAISE "^C" IF CtrlC() = TRUE

PROC main() HANDLE

	DEF spec     = NIL : PTR TO filterspecification,
	    input    = NIL : PTR TO oss_input,
	    output   = NIL : PTR TO oss_output,
	    splane   = NIL : PTR TO splane,
	    zplane   = NIL : PTR TO zplane,
	    filter   = NIL : PTR TO filter,
	    fxfilter = NIL : PTR TO fxfilter,
	    x, f, lf, hf, count = 0
#ifdef PLOT
	DEF draws    = NIL : PTR TO drawplane,
	    drawz    = NIL : PTR TO drawplane,
	    drawf    = NIL : PTR TO drawfreq
#endif

	oss_init()

	NEW spec.cli()

	NEW input.oss_input(spec.from)
	IF spec.rate = RATE_UNKNOWN THEN spec.rate := input.info.rate !

->	NEW output.oss_output(spec.to, spec.volume)
	output := setupoutput(spec.from, spec.to, input, spec.volume)

	x := spec.type
	SELECT 5 OF x

	CASE FILTERTYPE_BUTTERWORTH,
	     FILTERTYPE_CHEBYSHEV,
	     FILTERTYPE_BESSEL
		SELECT 5 OF x
		CASE FILTERTYPE_BUTTERWORTH;NEW splane.butterworth(spec, spec.order)
		CASE FILTERTYPE_CHEBYSHEV;  NEW splane.chebyshev  (spec, spec.order, spec.ripple)
		CASE FILTERTYPE_BESSEL;     NEW splane.bessel     (spec, spec.order)
		ENDSELECT
		x := spec.mode
		SELECT x
		CASE FILTERMODE_BILINEAR;   f := prebilinear(spec.freq,  spec.rate);   lf := prebilinear(spec.lfreq, spec.rate);   hf := prebilinear(spec.hfreq, spec.rate)
		CASE FILTERMODE_MATCHEDZ;   f := prematchedz(spec.freq,  spec.rate);   lf := prematchedz(spec.lfreq, spec.rate);   hf := prematchedz(spec.hfreq, spec.rate)
		DEFAULT;                    Throw("bug", 'main.splane.mode')
		ENDSELECT
		x := spec.shape
		SELECT x
		CASE FILTERSHAPE_LOWPASS;   splane.lowpass(f)
		CASE FILTERSHAPE_HIGHPASS;  splane.highpass(f)
		CASE FILTERSHAPE_BANDPASS;  splane.bandpass(lf, hf)
		CASE FILTERSHAPE_BANDSTOP;  splane.bandstop(lf, hf)
		DEFAULT;                    Throw("bug", 'main.splane.shape')
		ENDSELECT
		x := spec.mode
		SELECT x
		CASE FILTERMODE_BILINEAR;   NEW zplane.bilinear(splane)
		CASE FILTERMODE_MATCHEDZ;   NEW zplane.matchedz(splane)
		DEFAULT;                    Throw("bug", 'main.splane.mode')
		ENDSELECT
        ->easyguiA('s-plane', [PLUGIN, 0, NEW draws.drawsplane(splane)])

	CASE FILTERTYPE_RESONATOR
		f := preresonator(spec.freq, spec.rate)
		x := spec.shape
		SELECT x
		CASE FILTERSHAPE_BANDPASS;  NEW zplane.bandpass(spec, f, spec.qfactor)
		CASE FILTERSHAPE_ALLPASS;   NEW zplane.allpass (spec, f, spec.qfactor)
		CASE FILTERSHAPE_NOTCH;     NEW zplane.notch   (spec, f, spec.qfactor)
		DEFAULT;                    Throw("bug", 'main.resonator.shape')
		ENDSELECT

	DEFAULT;                        Throw("bug", 'main.type')
	ENDSELECT
#ifdef PLOT
	easyguiA('z-plane', [PLUGIN, 0, NEW drawz.drawzplane(zplane)])
#endif
	zplane.compile()

#ifdef PLOT
	x := spec.shape
	SELECT 7 OF x
	CASE FILTERSHAPE_LOWPASS,
	     FILTERSHAPE_HIGHPASS,
	     FILTERSHAPE_ALLPASS,
	     FILTERSHAPE_NOTCH
		easyguiA('frequency', [PLUGIN, 0, NEW drawf.drawzfreq(zplane, ! spec.freq / spec.rate)])
	CASE FILTERSHAPE_BANDSTOP
		easyguiA('frequency', [PLUGIN, 0, NEW drawf.drawzfreq(zplane, ! spec.lfreq / spec.rate, ! spec.hfreq / spec.rate)])
	CASE FILTERSHAPE_BANDPASS
		IF spec.type = FILTERTYPE_RESONATOR
			easyguiA('frequency', [PLUGIN, 0, NEW drawf.drawzfreq(zplane, ! spec.freq / spate)])
		ELSE
			easyguiA('frequency', [PLUGIN, 0, NEW drawf.drawzfreq(zplane, ! spec.lfreq / spec.rate, ! spec.hfreq / spec.rate)])
		ENDIF
	ENDSELECT
#endif

	NEW filter.filter(zplane)

	NEW fxfilter.fxfilter(input, output, filter)

	WHILE fxfilter.process() DO CtrlC() BUT count++

EXCEPT DO

	IF count THEN WriteF('\d samples processed\n', count)

	END fxfilter
	END filter
	END zplane
	END splane
	END output
	END input
	END spec
#ifdef PLOT
	END draws
	END drawz
	END drawf
#endif

	oss_cleanup()

	printerror(exception, exceptioninfo)

ENDPROC IF exception THEN 5 ELSE 0

/*-------------------------------------------------------------------------*/

-> catch exceptions from oss_output
PROC setupoutput(from, to, in : PTR TO oss_input, volume) HANDLE
	DEF out = NIL : PTR TO oss_output
	NEW out.oss_output(to, volume)
	IF from <> to
		IF out.info.length < in.info.length
			END out
			oss('sa_changesize \d', in.info.length)
			NEW out.oss_output(to, volume)
		ENDIF
	ENDIF
EXCEPT
	IF exception = "oss"
		END out
		oss('sa_changesize \d', in.info.length)
		NEW out.oss_output(to, volume)
	ELSE
		ReThrow()
	ENDIF
ENDPROC out

/*-------------------------------------------------------------------------*/
