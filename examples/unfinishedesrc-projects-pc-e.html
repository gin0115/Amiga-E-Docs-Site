PROC cursor(times,xpos,ypos)
DEF x

FOR x:=1 TO times
    Move(scr.rastport,xpos,ypos)
    Text(scr.rastport,'_',1)
    Delay(15)
    Move(scr.rastport,xpos,ypos)
    Text(scr.rastport,' ',1)
    Delay(15)
ENDFOR
ENDPROC

PROC prompt(xpos,ypos)

ENDPROC
