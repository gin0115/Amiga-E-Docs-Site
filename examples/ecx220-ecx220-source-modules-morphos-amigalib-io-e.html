PROC beginIO(ioreq)
  LWZ R0, .device(R3:io)
  STW R0, REG_A6 -> base
  STW R3, REG_A1 -> ioreq
  LWZ R0, .emulcalldirectos(R2:emulhandle)
  MTSPR 9, R0
  ADDI R3, R0, -30   -> offset
  BCCTRL 20, 0 -> call function
ENDPROC NIL -> no return value
