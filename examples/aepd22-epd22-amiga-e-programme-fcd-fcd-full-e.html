PROC setdir(what) HANDLE
DEF cmppath[LEN]:STRING,check=1,gefunden=0,num,i
num:=Long(args[ARG_NUM])     /* scheint ein Bug in E zu sein */
filehandle:=Open(path:=':fcd.file',MODE_OLDFILE);
address:=New(fs:=FileLength(path))
IF Read(filehandle,address,fs)<>fs THEN Raise(NOREAD)
tempadr:=address+4; LowerStr(args[ARG_PATH])
GetCurrentDirName(tpath,LEN)
StrCopy(cmppath,tpath+(i:=InStr(tpath,':',0)+1),ALL); StrCopy(volume,tpath,i)
IF what=SET_PATH
  i:= args[ARG_PATH]
  MOVE.L i,D1; MOVE.L tokenpath,D2; MOVE.L #TOKENLEN,D3
  MOVE.L dosbase,A6; JSR -$3c6(A6); MOVE.L D0,i       /* ParsePatternNoCase() */
  IF i=-1 THEN Raise(IoErr())
  IF args[ARG_SHOW]=-1
    i:=1
    WHILE check>0
      CtrlC(); StrCopy(path,'',ALL); INC gefunden
      IF getpath()=FALSE THEN check:=0 ELSE WriteF('\r\d[5] \r\d[5] \s\n',
                                                    i++,numdirs,path)
    ENDWHILE
    IF gefunden=1 THEN WriteF('nothing appropiate found\n')
  ELSE
    WHILE check>0       /* DEFAULT: check=1, gefunden=0 */
        StrCopy(path,'',ALL)
        IF getpath()=FALSE
          check:=0; tempadr:=address+4
        ELSE
          gefunden:=1; IF StrCmp(path,cmppath,ALL)=TRUE THEN check:=-1
        ENDIF
    ENDWHILE
    IF gefunden=0 THEN Raise(NODIR)
    IF check=1                              /* kein passendes Dir gefunden? */
      tempadr:=address+4; get_cycledir()
    ELSE
      get_cycledir()                        /* ansonsten nächstes Dir holen */
    ENDIF
  ENDIF
ELSE
  IF num--<0 THEN Raise(WRONGVAL)
  FOR i:=1 TO num
    tempadr:=Char(tempadr)+tempadr+5
    IF ^tempadr=0 THEN Raise(WRONGVAL)
  ENDFOR
  StrCopy(path,volume,ALL); makepath(tempadr)
ENDIF
IF args[ARG_SHOW]<>-1
  UnLock(CurrentDir(Lock(path,-2))); SetCurrentDirName(path)
ENDIF
Close(filehandle)
EXCEPT
  SELECT exception
    CASE BREAK;    WriteF('***break\n')
    CASE NOMEM;    WriteF('not enough memory to load \a:fcd.file\a\n')
    CASE NOOPEN;   WriteF('couldn\at open \a\s\a\n\s',
                           path,'try: \afcd create <volume>\a\n')
    CASE NOLOCK;   WriteF('Couldn\at lock to: \a\s\a\n'+
                          'brainfile probably not entirely valid\n'+
                          'type: \afcd create \s\a to update brainfile\n',path,volume)
    CASE NOREAD;   WriteF('error reading file: \a\s\a\n',path)
    CASE NODIR;    WriteF('\a\s\a doesn\at exist\n',args[ARG_PATH])
    CASE NOSETDIR; WriteF('couldn\at set dirstring; SetCurrentDirName() failed\n')
    CASE WRONGVAL; WriteF('wrong value\n')
    DEFAULT;       Raise(exception)
  ENDSELECT
  IF filehandle<>0 THEN Close(filehandle)
ENDPROC


PROC get_cycledir()
DEF x,z
z:=Long(args[ARG_CYCLE]); IF z=1 THEN z:=0 ELSE IF z THEN DEC z
FOR x:=0 TO z
  IF getpath()=FALSE
    tempadr:=address+4; getpath()
  ENDIF
ENDFOR
StrCopy(tpath,path,ALL); StrCopy(path,volume,ALL); StrAdd(path,tpath,ALL)
ENDPROC


PROC getpath()
DEF x,gefunden=0
IF Long(tempadr)=0 THEN RETURN FALSE
REPEAT
   StrCopy(path,'',ALL)
   INC numdirs; makepath(tempadr); tempadr:= tempadr+Char(tempadr)+5
   MOVE.L tokenpath,D1; MOVE.L path,D2
   MOVE.L dosbase,A6; JSR -$3cc(A6); MOVE.L D0,x
   IF x=0; IF x:= IoErr() THEN Raise(x); ELSE; gefunden:=1; ENDIF
UNTIL (Long(tempadr)=0) OR gefunden
IF gefunden THEN RETURN TRUE
ENDPROC FALSE


PROC makepath(adr:PTR TO CHAR)
IF ^adr
  dummy:= adr[]+adr; makepath(dummy-Long(dummy+1)+1); AddPart(path,adr+1,LEN)
ENDIF
ENDPROC


PROC checkdirtree() HANDLE
DEF l=0,info:fileinfoblock,lokal=0,ss,bb=0
l:=Lock(path,-2); Examine(l,info)
WHILE (ExNext(l,info))<>0
 CtrlC(); FreeStack()
 IF info.direntrytype>0
   INC diranz; ss:=StrLen(info.filename)
   lokal:=lokal+ss+global+1; bb:=bb+ss+5; global:=5+ss
   FputC(filehandle,ss); Fwrite(filehandle,info.filename,ss,1)
   Fwrite(filehandle,{lokal},4,1)
   lokal:=lokal+4
   SetStr(path,LEN); AddPart(path,info.filename,LEN)
   IF args[ARG_SHOW] THEN WriteF('\s\n',path)
   IF checkdirtree()=FALSE; UnLock(l); RETURN FALSE; ENDIF
   SetStr(path,PathPart(path)-path)
   bb:=bb+global
 ELSE; INC fileanz; volumefull:=volumefull+info.size; ENDIF
ENDWHILE
global:=bb; UnLock(l); RETURN TRUE
EXCEPT
  SELECT exception
    CASE NOLOCK;    WriteF('couldn\at lock to \a\s\a\n',path)
    CASE NOEXAMINE; WriteF('couldn\at examine \a\s\a\n',path)
    CASE NOFPUTC;   UnLock(l); Raise(NOFPUTC)
    CASE NOFWRITE;  UnLock(l); Raise(NOFWRITE)
    CASE BREAK;     UnLock(l); Raise(BREAK)
    CASE NOSTACK;   UnLock(l); Raise(NOSTACK)
    DEFAULT;        PrintFault(IoErr(),'DOS error')
  ENDSELECT
  IF l<>0 THEN UnLock(l)
  RETURN FALSE
ENDPROC
