EXPORT	PROC	__planar2chunky(bm:PTR TO bitmap,chunky,szer,wys)
DEF	p2c:p2cstruct

	p2c.bmap:=bm
	p2c.chunkybuffer:=chunky
	p2c.startx:=0
	p2c.starty:=0
	p2c.width:=szer
	p2c.height:=wys

	planarToChunkyAsm(p2c)
ENDPROC
