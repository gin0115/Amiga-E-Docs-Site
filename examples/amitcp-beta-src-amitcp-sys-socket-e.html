PROC cmsg_nxthdr(mhdr:PTR TO msghdr, cmsg:PTR TO cmsghdr) IS
  IF (cmsg+cmsg.len+SIZEOF cmsghdr) > (mhdr.control+mhdr.controllen) THEN
    NIL ELSE (cmsg+ALIGN(cmsg.len))
