   the same meaning as for sigprocmask(2). */

->PROC pthread_sigmask(__how:VALUE, __newmask:PTR TO __sigset_t, __oldmask:PTR TO __sigset_t) IS NATIVE {pthread_sigmask( (int) } __how {,} __newmask {,} __oldmask {)} ENDNATIVE !!VALUE
