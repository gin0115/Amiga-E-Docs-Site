PROC exceptionReport(pname)
DEF str[5]:ARRAY OF CHAR,str2:PTR TO CHAR,x
  IF exception
    PutQuad(str!!VALUE!!PTR TO QUAD,exception)
    str[4]:=0
    str2:=str
    WHILE str2[]=0 DO str2++

    FOR x:=0 TO 3
      IF (str[x]<>0) AND ((str[x]<32) OR (str[x]>127)) THEN str2:=-1 !!VALUE!!PTR TO CHAR
    ENDFOR

    IF exception<1000
      Print('\s: Exception #\d recieved!\n',pname,exception)
    ELSE IF str2=-1
      Print('\s: Exception $\z\h[8]\n recieved!\n',pname,exception)
    ELSE
      Print('\s: Exception "\s" recieved!\n',pname,str2)
    ENDIF
  ENDIF
ENDPROC
