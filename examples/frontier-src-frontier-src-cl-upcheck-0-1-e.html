/* RAW-Uploadchecker für Connectline 5.0 (not released because i`m not programming for connectline anymore!)
 *
 *
 * Features:
 *
 * - Packererkennung (mittels der 10 byte erkennung)
 * - File_ID.Diz kann benutzt werden, falls kein File_ID.Diz vorhanden, kann
 *   auf Wunsch des Uploaders sein eigener zum File eingegebener Kommentar
 *   als File_ID.Diz ans Archiv angehängt werden. (Kommt in CL-Version...)
 * - Erkennt, ob ein File schon existiert (Kommt in CL-Version...)
 * - Kann ein File nicht erkannt werden, so wird dieses per EXTERNEM Viren-
 *   Check überprüft ( JEDER Virenchecker wird unterstützt, sofern dieser
 *   das File, welches als Argument übergeben wurde auch korrekt überprüft...)
 *   Übergeben wird der Pfad und der Filename des Archives/Files.
 * 
 */

PROC checkIDfile()
DEF file, buf[20]:STRING, id[60]:STRING, charbuf[1]:STRING, oldid[60]:STRING, ausgabe[11]:STRING
 IF file:= Open('sys:prefs/env-archive/cl_upcheck.prf',OLDFILE)

  Read(file,buf,4)              -> FORM
  Read(file,buf,4)              -> 4 leerbytes
  Read(file,buf,8)              -> UPCHECKP
  Read(file,buf,4)              -> 1 leerbyte, 1 @, 3 leerbytes
  Read(file,buf,11)             -> Version:x.x
   StrCopy(ausgabe,buf,11)
    WriteF(' \n Lade Externes ID-File \s \n\n',ausgabe)
  Read(file,buf,11)             -> 1 leerbyte, 1 @, 4 leerbytes, BODY, 1 leerbyte

LOOP
 MidStr(oldid,id,3,57)          -> Alte ID sichern...
  StrCopy(id,'',ALL)           -> ID leeren...

REPEAT
   StrCopy(charbuf,' ',1)       -> zeichenbuffer leeren...
    Read(file,charbuf,1)        -> 1 Zeichen lesen...
     StrAdd(id,charbuf)         -> Dieses Zeichen an ID adden...
UNTIL StrCmp('\n',charbuf,1)=TRUE 

 IF StrCmp(header,id,4)=TRUE THEN JUMP fileerkannt
  IF CtrlC() OR StrCmp(id,'#->ENDE',7)=TRUE 
   type:=NONE
    JUMP nichterkannt
  ENDIF
ENDLOOP

fileerkannt:
type:=EXTERN
 StrCopy(fileid,oldid,60)

nichterkannt:
  Close(file)
ELSE
 WriteF(' Konnte die Prefs-Datei nicht finden!\n')
ENDIF
ENDPROC
