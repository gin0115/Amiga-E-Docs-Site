PROC arr_InitArray(arr_x,arr_y,arr_z,arr_type) OF array3

/* - wie `InitArray() OF array2' -- */

  IF (arr_x < 1) OR (arr_y < 1) OR (arr_z < 1) THEN RETURN

  IF (arr_type = ARR_BYTE) OR 
     (arr_type = ARR_INT)  OR
     (arr_type = ARR_LONG) 

    self.memory  := NIL       -> kein Speicher alloziert   
    self.protect := FALSE     -> Reihung besteht noch nicht
    self.unit    := arr_type  -> Größe einer Zelle         
    self.maxx    := arr_x     -> Dimensionen der Reihung   
    self.maxy    := arr_y
    self.maxz    := arr_z

    -> Speicherverbrauch in Bytes
    self.length  := arr_x * arr_y * arr_z * arr_type

    self.arr_CreateArray()    -> Speicher wird sofort reserviert

  ENDIF

ENDPROC


PROC arr_InitArray(arr_anzahl) OF booleans
DEF boo_temp,boo_ch

  -> Ein Feld muß mindestens vorhanden sein -- */
  IF arr_anzahl < 1 THEN RETURN
 
  self.memory        := NIL
  boo_temp,boo_ch    := Mod(arr_anzahl,8)  -> ein Byte kann 8 Zellen fassen
  IF boo_temp > 0
    boo_ch := boo_ch + 1             
  ENDIF
  self.length        := boo_ch
  self.indices       := arr_anzahl
  self.protect       := FALSE  

  self.arr_CreateArray()  -> Speicher wird reserviert

ENDPROC

PROC arr_CreateArray() OF array2

/* -- Erstellt das entsprechende Feld und setzt im Erfolgsfalle -- *
 * -- ein Schutzflag.                                           -- */

  IF Not(self.arr_ArrayExist()) /* -- Array noch nicht kreiert -- */
    self.memory  := AllocMem(self.length,MEMF_PUBLIC)
    IF self.memory <> NIL       /* -- Speicher vorhanden       -- */
      self.protect := TRUE      /* -- Schreibschutz            -- */
    ENDIF
  ENDIF

ENDPROC

PROC arr_SetCell(arr_x,arr_y,arr_value) OF array2

/* -- Das Element auf Position `arr_x|arr_y' erhält den Wert `arr_value' -- */

DEF arr_mem

  IF self.arr_ArrayExist() = TRUE     -> Is Speicher da ? 

    -> falls Irgendjemand Shit-Parameter benutzt
    IF (arr_x >= self.maxx) OR (arr_y >= self.maxy) THEN RETURN FALSE
    IF (arr_x < 0) OR (arr_y < 0) THEN RETURN FALSE

    -> Berechnung der Position im Speicher
    arr_mem := arr_y * self.maxx + arr_x
    arr_mem := arr_mem * self.unit + self.memory

    -> Der jeweilige Wert wird in der korrekten Größe geschrieben
    IF self.unit = ARR_BYTE
      PutChar(arr_mem,arr_value)
    ELSEIF self.unit = ARR_INT
      PutInt(arr_mem,arr_value)
    ELSE
      PutLong(arr_mem,arr_value)
    ENDIF

  ENDIF

ENDPROC TRUE


PROC arr_SetCell(arr_x,arr_y,arr_z,arr_value) OF array3

/* -- wie `arr_SetCell() OF array2' mit dem Unterschied, daß -- *
 * -- hier eine Koordinate für die 3.te Dimension angegeben  -- *
 * -- wird:  `arr_x|arr_y|arr_z'.                            -- */

DEF arr_mem

  IF self.arr_ArrayExist() = TRUE

    -> Sicherheitskontrolle ;-)
    IF (arr_x >= self.maxx) OR 
       (arr_y >= self.maxy) OR
       (arr_z >= self.maxz) THEN RETURN FALSE
    IF (arr_x < 0) OR (arr_y < 0) OR (arr_z < 0) THEN RETURN FALSE

    -> Berechnung der Position im Speicher
    arr_mem := arr_y * self.maxx + arr_x
    arr_mem := arr_z * self.maxx * self.maxy + arr_mem
    arr_mem := arr_mem * self.unit + self.memory

    -> Wert schreiben
    IF self.unit = ARR_BYTE
      PutChar(arr_mem,arr_value)
    ELSEIF self.unit = ARR_INT
      PutInt(arr_mem,arr_value)
    ELSE
      PutLong(arr_mem,arr_value)
    ENDIF

  ENDIF

ENDPROC TRUE


PROC arr_SetCell(arr_index,arr_val) OF booleans
DEF arr_temp,arr_byte,arr_ch

  -> Auf die Parameter aufpassen
  IF (arr_index < 0) OR (arr_index >= self.indices) THEN RETURN FALSE
  IF Not(self.arr_ArrayExist()) THEN RETURN FALSE
 
  arr_temp,arr_byte := Mod(arr_index,8)           -> Position und Index
  arr_ch            := Char(self.memory+arr_byte) -> Byte lesen

  MOVE.L                    D2,-(A7)   -> D2: auf Stack retten
  MOVE.L                arr_ch,D0      -> D0: betreffendes Byte
  MOVE.L              arr_temp,D1      -> D1: Index des Bits
  MOVE.L               arr_val,D2      -> D2: Wert (TRUE|FALSE)
  CMP.L             #$FFFFFFFF,D2      -> wenn D2 = TRUE, dann
  BNE.S                 arr_next0      -> springe nicht und
  BSET.L                    D1,D0      -> setze Bit
  BRA.S                 arr_next1      -> ENDE

arr_next0:
  BCLR.L                    D1,D0      -> andernfalls lösche Bit

arr_next1:
  MOVE.L                    D0,arr_ch  -> Wert in die Variable 
  MOVE.L                 (A7)+,D2      -> D2: vom Stack restaurieren

  PutChar(self.memory+arr_byte,arr_ch) -> Byte zurückschreiben

ENDPROC TRUE


PROC arr_SetCell(arr_index,arr_val) OF boolong
DEF arr_temp

  IF (arr_index < 0) OR (arr_index > 31) THEN RETURN FALSE

  arr_temp := self.array

  MOVE.L                    D2,-(A7)     -> D2: auf Stack retten
  MOVE.L             arr_index,D1        -> D1: Index des Elementes
  MOVE.L              arr_temp,D0        -> D0: betreffende Reihung
  MOVE.L               arr_val,D2        -> D2: Zu speichernder Wert
  CMP.L             #$FFFFFFFF,D2        -> wenn D2 = TRUE, dann
  BNE.S                 arr_next2        -> springe nicht und
  BSET.L                    D1,D0        -> setze Bit
  BRA.S                 arr_next3        -> ENDE

arr_next2:
  BCLR.L                    D1,D0        -> andernfalls lösche Bit

arr_next3:
  MOVE.L                    D0,arr_temp  -> Wert in die Variable 
  MOVE.L                 (A7)+,D2        -> D2: vom Stack restaurieren

  self.array := arr_temp

ENDPROC TRUE

PROC arr_GetCell(arr_x,arr_y) OF array2

/* -- Diese Methode liefert den Wert zurück, der an der Stelle -- *
 * -- `arr_x|arr_y' steht.                                     -- */

DEF arr_mem

  IF self.arr_ArrayExist() -> Speicherbereich muß existieren

    -> Parameter überprüfen
    IF (arr_x >= self.maxx) OR (arr_y >= self.maxy) THEN RETURN 0,TRUE
    IF (arr_x < 0) OR (arr_y < 0) THEN RETURN 0,TRUE

    -> ??? ;-)
    arr_mem := arr_y * self.maxx + arr_x
    arr_mem := arr_mem * self.unit + self.memory

    -> Wert der Zelle als Funktionswert liefern
    IF self.unit = ARR_BYTE
      RETURN Char(arr_mem),FALSE
    ELSEIF self.unit = ARR_INT
      RETURN Int(arr_mem),FALSE
    ELSE
      RETURN Long(arr_mem),FALSE
    ENDIF

  ENDIF

ENDPROC


PROC arr_GetCell(arr_x,arr_y,arr_z) OF array3

/* -- wie `arr_GetCell() OF array2', jedoch mit drittem Parameter -- *
 * -- Kommentierung erübrigt sich !!!                             -- */

DEF arr_mem

  IF self.arr_ArrayExist()

    IF (arr_x >= self.maxx) OR 
       (arr_y >= self.maxy) OR
       (arr_z >= self.maxz) THEN RETURN 0,TRUE

    IF (arr_x < 0) OR 
       (arr_y < 0) OR
       (arr_z < 0) THEN RETURN 0,TRUE

    arr_mem := arr_y * self.maxx + arr_x
    arr_mem := arr_z * self.maxx * self.maxy + arr_mem
    arr_mem := arr_mem * self.unit + self.memory

    IF self.unit = ARR_BYTE
      RETURN Char(arr_mem),FALSE
    ELSEIF self.unit = ARR_INT
      RETURN Int(arr_mem),FALSE
    ELSE
      RETURN Long(arr_mem),FALSE
    ENDIF

  ENDIF

ENDPROC

PROC arr_UsedMem() OF array2
DEF arr_used

  IF self.arr_ArrayExist()
    arr_used := self.arr_MaxX() * self.arr_MaxY() * self.arr_CellSize()
    arr_used := arr_used + SIZEOF array2
  ELSE
    arr_used := SIZEOF array2
  ENDIF

PROC arr_Copy() OF array2
DEF arr_array2 : PTR TO array2

  NEW arr_array2.arr_InitArray(self.maxx,self.maxy,self.unit)
  IF arr_array2.arr_ArrayExist() = TRUE

    -> Das Kopieren geht des ganzen Speicherbereichs geht
    -> schneller als das Auslesen und Schreiben einzelner
    -> Speicherzellen. 
    CopyMem(self.memory,arr_array2.memory,self.length)

  ELSE
    RETURN NIL
  ENDIF

ENDPROC arr_array2


PROC arr_Copy() OF array3
DEF arr_array3 : PTR TO array3

  NEW arr_array3.arr_InitArray(self.maxx,self.maxy,self.maxz,self.unit)
  IF arr_array3.arr_ArrayExist() = TRUE

  ELSE

  ENDIF
ENDPROC arr_array3
