MODULE 'dos/dos','dos/datetime'

->> DEFINITIONS GLOBALES

DEF dat:PTR TO datetime
DEF p
DEF strdate[LEN_DATSTRING]:STRING
DEF strday[LEN_DATSTRING]:STRING
DEF strtime[LEN_DATSTRING]:STRING

-><
->> main()
PROC main()
    dat:=New(SIZEOF datetime)
    p:=New(SIZEOF datestamp)
    dat.stamp:days:=0
    dat.stamp.minutes:=7
    dat.stamp.tick:=7*60*50
    dat.format:=FORMAT_CDN
    dat.flags:=0
    dat.strdate:=strdate
    dat.strday:=strday
    dat.strtime:=strtime
    DateToStr(dat)
    WriteF('\s',strtime)
    Dispose(dat)
ENDPROC
