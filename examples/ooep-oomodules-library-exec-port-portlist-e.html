DEF signal,
    port:PTR TO port,
    index=-1

  IF self.tail = 0 THEN RETURN

  signal := self.wait()

  REPEAT

    INC index
    port := self.val[index]

  UNTIL (signal AND port.getSignalMask())

->  WriteF('Message arrived at port \d.\n', port)

  RETURN port.getMsg(mode), self.key[index], port

ENDPROC
