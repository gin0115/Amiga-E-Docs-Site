PROC wait4message(win:PTR TO window)
  DEF mes:PTR TO intuimessage,type
  REPEAT
    type:=0
    IF mes:=Gt_GetIMsg(win.userport)
      type:=mes.class
      IF type=IDCMP_GADGETUP
        infos:=mes.iaddress
        a := infos.gadgetid
      ELSEIF type=IDCMP_REFRESHWINDOW
        Gt_BeginRefresh(win)
        Gt_EndRefresh(win,TRUE)
        type:=0
      ELSEIF type<>IDCMP_CLOSEWINDOW  /* remove these if you like */
        type:=0
      ENDIF
      Gt_ReplyIMsg(mes)
    ELSE
      WaitPort(win.userport)
    ENDIF
  UNTIL type
ENDPROC type

PROC main()

  VOID '$VER: Playerek ver.0.1006ß (06.10.96) by Leo'

  IF reporterr(setup())=0
    reporterr(openmainwindow())
    req:=RtAllocRequestA(RT_FILEREQ,NIL)
    WriteF('\n')
    name:=arg
    len := StrLen(name)
    IF len=0
      loadreq()
    ELSE
      load()
    ENDIF


-> main

    REPEAT
      type := wait4message(mainwnd)

      SELECT type
        CASE IDCMP_CLOSEWINDOW
          eject()
          BRA y

        CASE IDCMP_GADGETUP
          SELECT a -> infos

            CASE GPLAY
              IF module
                PtStop(module)
                play()
              ELSE
                loadreq()
              ENDIF

            CASE GPAUSE
              IF pauza
                PtResume(module)
                pauza := FALSE
              ELSE
                PtPause(module)
                pauza := TRUE
              ENDIF

            CASE GEJECT
              IF module
                eject()
              ENDIF
            CASE GNEXT
              PtSetPos(module,PtSongPos(module)+1)
            CASE GPREV
              PtSetPos(module,PtSongPos(module)-1)
          ENDSELECT
      ENDSELECT
      IF CtrlC() THEN BRA y
    UNTIL type = IDCMP_CLOSEWINDOW

    y: closemainwindow()
  ENDIF
  exit()

ENDPROC

PROC play()

    PtPlay(module)
    PtPause(module)
    PtResume(module)

ENDPROC

PROC eject()

  IF module
    PtStop(module)
    PtFreeMod(module)
    FreeMem(mod,len)
    mod := NIL
    len := NIL
    module := NIL
    StrCopy(tytul,'  Brak moduîu w pamiëci')
    autor[0] := 0
    Gt_SetGadgetAttrsA(gpa,mainwnd,NIL,[GA_DISABLED,TRUE,NIL])
    Gt_SetGadgetAttrsA(gej,mainwnd,NIL,[GA_DISABLED,TRUE,NIL])
    Gt_SetGadgetAttrsA(gne,mainwnd,NIL,[GA_DISABLED,TRUE,NIL])
    Gt_SetGadgetAttrsA(gpr,mainwnd,NIL,[GA_DISABLED,TRUE,NIL])
    title()
  ENDIF

ENDPROC

PROC loadreq()

DEF temp[120]:STRING,
    test[2]:STRING,
    fname[120]:STRING,
    dir

  test[0] := 0
  fname[0] := 0
  temp[0] := 0
  IF mod THEN FreeMem(mod,len)
  IF RtFileRequestA(req,fname,'Wybierz moduî',0)
    MOVE.L req,A0
    MOVE.L 16(A0),dir
    StrCopy(temp,dir,ALL)
    RightStr(test,temp,1)
    IF StrCmp(test,'/',1)=FALSE AND StrCmp(test,':',1)=FALSE AND StrLen(dir)>0
      StrAdd(temp,'/',1)
    ENDIF
    StrAdd(temp,fname,ALL)
    name := temp
  ENDIF
  StrCopy(tytul,'Zaczekaj ladujë moduî: ')
  autor := fname
  title()
  autor[0] := 0
  load()
ENDPROC

PROC load()

DEF file,
    tmp,
    mk[4]:STRING

  file := Open(name,OLDFILE)
  IF file > NIL
    Seek(file,NIL,OFFSET_END)
    len := Seek(file,NIL,OFFSET_BEGINNING)
    mod := AllocMem(len,MEMF_CHIP OR MEMF_CLEAR)
    IF mod >NIL
      Read(file,mod,len)
      mk := (mod + 1080)
      tmp := InStr(mk,'M.K.')
      IF (InStr(mk,'M.K.') = 0) OR (InStr(mk,'M!K!') = 0)
        module := PtSetupMod(mod)
        StrCopy(tytul,mod,22)
        play()
        title()
        checkautor()
        Gt_SetGadgetAttrsA(gpa,mainwnd,NIL,[GA_DISABLED,FALSE,NIL])
        Gt_SetGadgetAttrsA(gej,mainwnd,NIL,[GA_DISABLED,FALSE,NIL])
        Gt_SetGadgetAttrsA(gne,mainwnd,NIL,[GA_DISABLED,FALSE,NIL])
        Gt_SetGadgetAttrsA(gpr,mainwnd,NIL,[GA_DISABLED,FALSE,NIL])
      ELSE
        StrCopy(tytul,'To nie jest modul')
        IF mod THEN FreeMem(mod,len)
        module := NIL
        mod := NIL
      ENDIF
    ELSE
      StrCopy(tytul,'Zabrakîo pamiëci')
      module := NIL
      mod := NIL
    ENDIF
  ELSEIF file = NIL
    StrCopy(tytul,'Nie mogë odczytaê pliku')
    module := NIL
    mod := NIL
  ENDIF -> IF Open
  title()
  IF file THEN Close(file)

ENDPROC
