NATIVE {DEVICES_TIMER_H} CONST

/* unit defintions */

NATIVE {UNIT_VBLANK}	CONST UNIT_VBLANK	= 1

NATIVE {UNIT_WAITECLOCK}	CONST UNIT_WAITECLOCK	= 4

    {ev_lo}	lo	:ULONG
ENDOBJECT

    {tr_node}	io	:io

ENDOBJECT
