/* Converts x from degrees to radians */

PROC dRad(x:PTR TO longreal,to=NIL:PTR TO longreal)
  DEF s:longreal,t:longreal
   
  dPi(t)
  dDouble(180.0,s)

  dDiv(t,s)
  dMul(t,x)
  IF to
    to.a:=t.a
    to.b:=t.b
    RETURN to
  ELSE
    x.a:=t.a
    x.b:=t.b
    RETURN x
  ENDIF
ENDPROC
