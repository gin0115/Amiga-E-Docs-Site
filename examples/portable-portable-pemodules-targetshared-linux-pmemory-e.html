MODULE 'target/unistd'	->{#include <unistd.h>}

PROC MemAvail() RETURNS sizeInBytes:BIGVALUE
	NATIVE {
	long pages = sysconf(_SC_PHYS_PAGES);
	long page_size = sysconf(_SC_PAGE_SIZE);
	return pages * page_size;
    } ENDNATIVE
ENDPROC
