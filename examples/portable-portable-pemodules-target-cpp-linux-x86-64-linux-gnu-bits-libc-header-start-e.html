/* This header is internal to glibc and should not be included outside
   of glibc headers.  Headers including it must define
   __GLIBC_INTERNAL_STARTING_HEADER_IMPLEMENTATION first.  This header
   cannot have multiple include guards because ISO C feature test
   macros depend on the definition of the macro when an affected
   header is included, not when the first system header is
   included.  */

/* ISO/IEC TR 24731-2:2010 defines the __STDC_WANT_LIB_EXT2__
   macro.  */
 ->NATIVE {__GLIBC_USE_LIB_EXT2} CONST __GLIBC_USE_LIB_EXT2 = 1

/* ISO/IEC TS 18661-1:2014 defines the __STDC_WANT_IEC_60559_BFP_EXT__
   macro.  Most but not all symbols enabled by that macro in TS
   18661-1 are enabled unconditionally in C2X; the symbols in Annex F
   still require that macro in C2X.  */
 ->NATIVE {__GLIBC_USE_IEC_60559_BFP_EXT} CONST __GLIBC_USE_IEC_60559_BFP_EXT = 1
 ->NATIVE {__GLIBC_USE_IEC_60559_BFP_EXT_C2X} CONST __GLIBC_USE_IEC_60559_BFP_EXT_C2X = 1

/* ISO/IEC TS 18661-4:2015 defines the
   __STDC_WANT_IEC_60559_FUNCS_EXT__ macro.  Other than the reduction
   functions, the symbols from this TS are enabled unconditionally in
   C2X.  */
 ->NATIVE {__GLIBC_USE_IEC_60559_FUNCS_EXT} CONST __GLIBC_USE_IEC_60559_FUNCS_EXT = 1
 ->NATIVE {__GLIBC_USE_IEC_60559_FUNCS_EXT_C2X} CONST __GLIBC_USE_IEC_60559_FUNCS_EXT_C2X = 1

/* ISO/IEC TS 18661-3:2015 defines the
   __STDC_WANT_IEC_60559_TYPES_EXT__ macro.  */
 ->NATIVE {__GLIBC_USE_IEC_60559_TYPES_EXT} CONST __GLIBC_USE_IEC_60559_TYPES_EXT = 1
